// ========================================== // 1. 최고 기록(Personal Best) 시스템 // ========================================== // 화면에 기록을 불러와서 업데이트하는 함수 function loadBestRecords() { // 전선 연결 (타임어택: 초 단위) const bestWiring = localStorage.getItem('best_wiring'); const wiringElem = document.getElementById('record-wiring'); if (wiringElem) { wiringElem.textContent = bestWiring ? `${parseFloat(bestWiring).toFixed(2)}초` : '기록 없음'; } // 한자 쓰기 const bestKanji = localStorage.getItem('best_kanji'); const kanjiElem = document.getElementById('record-kanji'); if (kanjiElem) { kanjiElem.textContent = bestKanji ? `${parseFloat(bestKanji).toFixed(2)}초` : '기록 없음'; } // 퓨즈 박스 const bestFuse = localStorage.getItem('best_fuse'); const fuseElem = document.getElementById('record-fuse'); if (fuseElem) { fuseElem.textContent = bestFuse ? `${parseFloat(bestFuse).toFixed(2)}초` : '기록 없음'; } } // 새로운 기록 달성 시 저장 및 Alert 표시 함수 function updateBestRecord(gameType, newTime) { const currentBest = localStorage.getItem(`best_${gameType}`); // 기존 기록이 없거나, 더 짧은 시간(신기록)을 달성한 경우 if (!currentBest || newTime < parseFloat(currentBest)) { localStorage.setItem(`best_${gameType}`, newTime); alert(`🎉 [${getGameName(gameType)}] 새로운 최고 기록 달성!\n기록: ${newTime.toFixed(2)}초`); loadBestRecords(); // UI 업데이트 return true; } else { alert(`클리어! 소요 시간: ${newTime.toFixed(2)}초\n(최고 기록: ${parseFloat(currentBest).toFixed(2)}초)`); return false; } } // 종목 키값을 한글 명칭으로 변환해주는 헬퍼 함수 function getGameName(gameType) { switch (gameType) { case 'wiring': return '전선 연결'; case 'kanji': return '한자 쓰기'; case 'fuse': return '퓨즈 박스'; default: return '미니게임'; } } // 특정 종목의 최고 기록을 초기화하고 싶을 때 사용하는 함수 (선택 사항) function resetRecord(gameType) { localStorage.removeItem(`best_${gameType}`); loadBestRecords(); } // ========================================== // 2. 타이머 기능 // ========================================== let startTime = 0; let timerInterval = null; function startTimer() { startTime = performance.now(); const timerDisplay = document.getElementById('timer-display'); if (timerInterval) clearInterval(timerInterval); timerInterval = setInterval(() => { const currentTime = performance.now(); const elapsedTime = (currentTime - startTime) / 1000; if (timerDisplay) { timerDisplay.textContent = `${elapsedTime.toFixed(2)}초`; } }, 10); } function stopTimer() { if (timerInterval) { clearInterval(timerInterval); timerInterval = null; } const endTime = performance.now(); const totalTime = (endTime - startTime) / 1000; return totalTime; // 소요 시간(초) 반환 } // ========================================== // 3. 각 미니게임 클리어 이벤트 연동 // ========================================== // 예시 1: 전선 연결 게임 완료 시 function completeWiringGame() { const clearTime = stopTimer(); updateBestRecord('wiring', clearTime); } // 예시 2: 한자 쓰기 게임 완료 시 function completeKanjiGame() { const clearTime = stopTimer(); updateBestRecord('kanji', clearTime); } // 예시 3: 퓨즈 박스 게임 완료 시 function completeFuseGame() { const clearTime = stopTimer(); updateBestRecord('fuse', clearTime); } // ========================================== // 4. 페이지 초기화 (DOM 로드 시) // ========================================== window.addEventListener('DOMContentLoaded', () => { // 저장된 최고 기록 UI 세팅 loadBestRecords(); });
